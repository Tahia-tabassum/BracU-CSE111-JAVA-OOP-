public class Company{
  public Employee [] info= new Employee[3];
  public int count;
  
  
  public void addEmployee( Employee n){
    if(count<3){
     info[count]=n;
     count++;
    System.out.println(n.getNm()+" has joined the company"); }
    else{
      System.out.println("No more vacancy"); }
  }
  
  public void removeEmployee( Employee n){
    int index=-1;
    for(int i=0; i<count; i++){
      if(info[i]==n){
        index=i;
        break;}
      }
    if(index!=-1){
      for(int i=index; i<count-1; i++){
        info[i]=info[i+1];}
      //info[count-1]=null;
      count--;
      System.out.println(n.getNm()+" has left the company");}
  }
  
  
  public void details(){
  System.out.println("Company Name: ABC Company\nTotal Employee: "+count);
  System.out.println("Fulltime Employees: ");
  for(int i=0; i<count; i++){
    Employee n= info[i];
    if("Fulltime".equals(n.getType())){
    System.out.println("Name:"+n.getNm()+", ID:"+n.getId());}
  }
  
   System.out.println("Part-time Employees: ");
  for(int i=0; i<count; i++){
    Employee n= this.info[i];
    if("Part-time".equals(n.getType())){
    System.out.println("Name:"+n.getNm()+", ID:"+n.getId());}
  }
}
}

public class Spaceship{
  public String name;
  public int capacity, count, total, difference;
  public Cargo[] names= new Cargo[100];
  
  public Spaceship(String n, int c){
    this.name=n;
    this.capacity=c;
  }
  
  public void loadCargo( Cargo c){
    if((total+c.getW())<capacity){
      names[count]=c;
      count++;
      total+=c.getW();
      
    }
    else{
      difference=(c.getW()-(capacity-total));
      System.out.println("Warning: Unable to load "+c.getItem()+" inside "+name+". Exceeds capacity by "+difference+".");
    }
  }
  
  public void displayDetails(){
  System.out.println("Spaceship Name: "+name+"\nCapacity: "+capacity+"\nCurrent cargo weight: "+total+"\nCargo: ");
  for(int i=0; i<count;i++){
   System.out.print(names[i].getItem()+" ");
  }
  }
}
