public class Product{
  public static String []nm= new String [3];
    public static int []qty= new int [3];
  public static int count;
  
  
  
  public Product(String n, int q){
    if(count<3){
    this.nm[count]=n;
    this.qty[count]=q;
    count++;
    }
    else{
      System.out.println("Storage is full! Cannot add "+n);}
  }
  
  
  public static void buy(String n, int q){
    for(int i=0;i<count; i++){
      if(nm[i].equals(n)){
        if(qty[i]>=q){
        qty[i]-=q;
        System.out.println("Product sold");}
      else{
        System.out.println("Quantity low");}
      return; }
    }
 
          System.out.println("Product not found");
    }
  
  
  public static void displayProducts(){
    System.out.println("=== Stored Products ===");
    for(int i=0; i< count; i++){
      System.out.println(nm[i]+" - Qty: "+ qty[i]);}
  }
}

    this.nm=n;

    System.out.println("ID: "+total+" , Name: "+this.nm+"\nGroup: "+this.grp+"\nHealth: "+health);
  }

   System.out.println("Total Characters: "+total);

      avg=(totalH*1.0/total);}
    else{
      avg=0;}

     System.out.println("Strongest Character: ");
     if(total==0){

    }

  public static void AddtoVault( Artifact x){
    if(count>=4){
      System.out.println("!!"+x.owner+" unsuccessful in adding artifact to the vault!!");
      return;}
    
    if(count<4){
      vault[count]=x;
      count++;
     System.out.println(x.owner+" added "+x.name+" successfully to the vault.");
    }
  }
    
    
  public int CalcPower(){
    total=0;
    for(int i=0; i<name.length(); i++){
      if(name.length()%2==0 && i%2==0){
        total+= ((int)(name.charAt(i)));}
        else if(name.length()%2!=0 && i%2!=0){
          total+=((int)(name.charAt(i))); }
                    }
      return total;  }          
    
 public void revealArtifact(){
   
    System.out.println(name+" added by "+owner+" has power of "+CalcPower());
 }

   this.name=n;

 public static String strongest(){
