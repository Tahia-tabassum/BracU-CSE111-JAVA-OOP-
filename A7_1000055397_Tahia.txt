public class Book extends Product{
  public String isbn, publisher;
  
  public Book(int i, String t, int p, String is, String pub){
    super(i,t,p);
    this.isbn=is;
    this.publisher=pub;
  }
  
  public String printDetail(){
    
    return super.getIdTitlePrice()+"\nISBN: "+isbn+" Publisher: "+publisher; }
}

public class CD extends Product{
  public String band, genre;
  public int duration;
  public CD(int i, String t, int p, String b, int d,String g){
    super(i,t,p);
    this.band=b;
    this.duration=d;
    this.genre=g;
  }
  
  public String printDetail(){
    
    return super.getIdTitlePrice()+"\nBand: "+band+" Duration: "+duration+"\nGenre: "+genre; }
}
