public class PlatinumCard extends CreditCard {

    
    public PlatinumCard(String ch, String a) {
        super(ch, a, 100);
    }

    
    public void spendCash(int amnt) {
        System.out.println("Previous Reward Points: " + rewardPoints);
        
        int earned = (amnt / 100) * 2;
        rewardPoints += earned;

        System.out.println("Reward points after spending " + amnt+ " taka: " + rewardPoints);
    }
}
