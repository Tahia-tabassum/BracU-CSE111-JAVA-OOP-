public class Assignment8{
  
   public static boolean isPrime( int n1 ){
     if(n1<=1){
       return false; }
     for(int i=2; i<= n1/2; i++){
       if(n1%i ==0){
         return false;
         }
     }
     return true;
     }
   
     public static boolean isPerfect( int n2 ){
         int sum=1;
          if(n2<=1){
       return false; }
       for(int i=2; i<=n2/2; i++){
         if(n2%i==0){
           sum=sum+i; }
       }
         if(sum==n2){
           return true;
         }
         return false;
       }
      
     
      public static int special_sum( int n3 ){
        int result=0;
        for(int i=1; i<=n3; i++){
          if(isPrime(i) || isPerfect(i) ){
            result+=i; }
        }
          return result;
           }
          
      
      public static void main(String[]args){
       boolean check= isPrime(7);
        System.out.println(check);
        boolean perfectCheck = isPerfect(6);
        System.out.println(perfectCheck);
        int result= special_sum(8);
        System.out.println(result);
      }
}

import java.util.Scanner;
public class Assignment8{
  
   public static void showDots( int dot){
     for(int i=1; i<=dot; i++){
     System.out.print("."); }
   }
   
     public static void show_palindrome( int p ){
         for(int i=1; i<=p; i++){
           System.out.print(i);}
         for(int j=p-1; j>=1; j--){
           System.out.print(j);}
     }
      
     
      public static void showDiamond( int d ){
        for(int i=1; i<=d; i++){
          showDots(d-i);
          show_palindrome(i);
          showDots(d-i);
          System.out.println();
        }
         for(int i=d-1; i>=1; i--){
          showDots(d-i);
          show_palindrome(i);
          showDots(d-i);
          System.out.println(); }
        
        }
          
      
      public static void main(String[]args){
        Scanner sc= new Scanner ( System.in) ;
        int dot= sc.nextInt();
       showDots(dot);
       int p= sc.nextInt();
       show_palindrome(p);
       int d= sc.nextInt();
       showDiamond(d);
      }
}
